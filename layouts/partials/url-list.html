{{- /* Normalize a frontmatter value into a LIST of url strings. Accepts: a plain string, a YAML list of strings, or a YAML list of image objects. Decap's `list` widget stores multi-upload images as [{url: "..."}, ...], while batch.md and legacy posts use plain strings. Always returns a slice of strings. Usage: {{ $imgs := partial "url-list.html" .Params.pinImage }} */ -}} {{- $v := . -}} {{- $out := slice -}} {{- if reflect.IsSlice $v -}} {{- range $v -}} {{- if reflect.IsMap . -}} {{- $s := "" -}} {{- if index . "url" }}{{- $s = index . "url" -}} {{- else if index . "image" }}{{- $s = index . "image" -}} {{- else if index . "src" }}{{- $s = index . "src" -}} {{- else if index . "path" }}{{- $s = index . "path" -}} {{- end -}} {{- if $s }}{{- $out = $out | append $s -}}{{- end -}} {{- else if . -}} {{- $out = $out | append . -}} {{- end -}} {{- end -}} {{- else if $v -}} {{- $out = $out | append $v -}} {{- end -}} {{- return $out -}}
