{{- /* Structured data (JSON-LD) for Google Rich Results. - Article (with cover image / published+modified dates / publisher) on pages - BreadcrumbList on every non-home page - WebSite on the home page Uses jsonify so titles/descriptions with quotes can't break the JSON. */ -}} {{- $seo := merge site.Params (hugo.Data.seo | default dict) -}} {{- $brand := $seo.brand | default site.Title -}} {{- $pageTitle := .Params.seoTitle | default .Title -}} {{- $desc := .Params.seoDescription | default .Description | default (plainify .Summary | truncate 160) | default $seo.description -}} {{ if .IsPage }} {{- $img := "" -}} {{- with .Params.cover }}{{ $img = partial "first-url.html" . }}{{ end -}} {{- if not $img }}{{ with .Params.pinImage }}{{ $img = partial "first-url.html" . }}{{ end }}{{ end -}} {{- $imgAbs := "" -}} {{- with $img }}{{ if strings.HasPrefix . "http" }}{{ $imgAbs = . }}{{ else }}{{ $imgAbs = printf "%s%s" $.Site.BaseURL (strings.TrimPrefix "/" .) }}{{ end }}{{ end -}} {{- $article := dict "@context" "https://schema.org" "@type" "Article" "headline" $pageTitle "description" $desc "author" (dict "@type" "Organization" "name" $brand) "publisher" (dict "@type" "Organization" "name" $brand) "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) -}} {{- with $imgAbs }}{{ $article = merge $article (dict "image" .) }}{{ end -}} {{- if not .Date.IsZero }}{{ $article = merge $article (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) }}{{ end -}} {{ end }} {{ if not .IsHome }} {{- $crumbs := slice -}} {{- $pos := 1 -}} {{- range .Ancestors.Reverse -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" $pos "name" .Title "item" .Permalink) -}} {{- $pos = add $pos 1 -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" $pos "name" $pageTitle "item" .Permalink) -}} {{- $breadcrumb := dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $crumbs -}} {{ end }} {{ if .IsHome }} {{ end }}
